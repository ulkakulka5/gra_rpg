146 private async void MoveCharacter(Image character, double newX, double newY, bool checkCollision)
147 {
148 if (isNavigating)
149 return;
150
151 if (MainGrid.Width <= 0 || MainGrid.Height <= 0)
152 {
153 character.TranslationX = newX;
154 character.TranslationY = newY;
155 return;
156 }
157
158 double maxX = MainGrid.Width - character.Width;
159 double maxY = MainGrid.Height - character.Height;
160
161 newX = Math.Max(0, Math.Min(newX, maxX));
162 newY = Math.Max(0, Math.Min(newY, maxY));
163
164 if (!checkCollision || !await IsBlocked(newX, newY, character.Width, character.Height))
165 {
166 character.TranslationX = newX;
167 character.TranslationY = newY;
168 }
169 }
