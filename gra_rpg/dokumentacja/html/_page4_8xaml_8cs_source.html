204 private async Task MoveCharacter(Image character, double newX, double newY, bool checkCollision)
205 {
206 if (isNavigating)
207 return;
208
209 if (MainGrid.Width <= 0 || MainGrid.Height <= 0)
210 {
211 character.TranslationX = newX;
212 character.TranslationY = newY;
213 return;
214 }
215
216 double maxX = MainGrid.Width - character.Width;
217 double maxY = MainGrid.Height - character.Height;
218
219 newX = Math.Max(0, Math.Min(newX, maxX));
220 newY = Math.Max(0, Math.Min(newY, maxY));
221
222 if (!checkCollision || !await IsBlocked(newX, newY, character.Width, character.Height))
223 {
224 // Animate movement for smoother walking. Duration scales with distance.
225 double dx = newX - character.TranslationX;
226 double dy = newY - character.TranslationY;
227 double dist = Math.Sqrt(dx * dx + dy * dy);
228 // duration between 60ms and 350ms depending on distance
229 uint duration = (uint)Math.Max(60, Math.Min(350, (int)(dist * 4)));
230
231 try
232 {
233 await character.TranslateTo(newX, newY, duration, Easing.CubicOut);
234 }
235 catch
236 {
237 // If animation is canceled or fails, fallback to direct placement
238 character.TranslationX = newX;
239 character.TranslationY = newY;
240 }
241 }
242 }
