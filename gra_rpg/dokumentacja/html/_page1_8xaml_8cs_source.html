133 private async void MoveCharacter(Image character, double newX, double newY, bool checkCollision)
134 {
135 if (isNavigating)
136 return;
137 if (MainGrid.Width <= 0 || MainGrid.Height <= 0)
138 {
139 character.TranslationX = newX;
140 character.TranslationY = newY;
141 return;
142 }
143
144 double maxX = MainGrid.Width - character.Width;
145 double maxY = MainGrid.Height - character.Height;
146
147 newX = Math.Max(0, Math.Min(newX, maxX));
148 newY = Math.Max(0, Math.Min(newY, maxY));
149
150 if (!checkCollision || !await IsBlocked(newX, newY, character.Width, character.Height))
151 {
152 character.TranslationX = newX;
153 character.TranslationY = newY;
154 }
155 }
