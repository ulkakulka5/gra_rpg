203 private async Task MoveCharacter(Image character, double newX, double newY, bool checkCollision)
204 {
205 if (isNavigating)
206 return;
207
208 if (MainGrid.Width <= 0 || MainGrid.Height <= 0)
209 {
210 character.TranslationX = newX;
211 character.TranslationY = newY;
212 return;
213 }
214
215 double maxX = MainGrid.Width - character.WidthRequest;
216 double maxY = MainGrid.Height - character.HeightRequest;
217
218 newX = Math.Max(0, Math.Min(newX, maxX));
219 newY = Math.Max(0, Math.Min(newY, maxY));
220
221 double charWidth = character.WidthRequest > 0 ? character.WidthRequest : 30;
222 double charHeight = character.HeightRequest > 0 ? character.HeightRequest : 30;
223
224 if (!checkCollision || !await IsBlocked(newX, newY, charWidth, charHeight))
225 {
226 // Animate movement for smoother walking. Duration scales with distance.
227 double dx = newX - character.TranslationX;
228 double dy = newY - character.TranslationY;
229 double dist = Math.Sqrt(dx * dx + dy * dy);
230 // duration between 60ms and 350ms depending on distance
231 uint duration = (uint)Math.Max(60, Math.Min(350, (int)(dist * 4)));
232
233 try
234 {
235 await character.TranslateTo(newX, newY, duration, Easing.CubicOut);
236 }
237 catch
238 {
239 // If animation is canceled or fails, fallback to direct placement
240 character.TranslationX = newX;
241 character.TranslationY = newY;
242 }
243 }
244 }
