151 private async void MoveCharacter(Image character, double newX, double newY, bool checkCollision)
152 {
153 if (isNavigating)
154 return;
155
156 if (MainGrid.Width <= 0 || MainGrid.Height <= 0)
157 {
158 character.TranslationX = newX;
159 character.TranslationY = newY;
160 return;
161 }
162
163 double maxX = MainGrid.Width - character.WidthRequest;
164 double maxY = MainGrid.Height - character.HeightRequest;
165
166 newX = Math.Max(0, Math.Min(newX, maxX));
167 newY = Math.Max(0, Math.Min(newY, maxY));
168
169 double charWidth = character.WidthRequest > 0 ? character.WidthRequest : 30;
170 double charHeight = character.HeightRequest > 0 ? character.HeightRequest : 30;
171
172 if (!checkCollision || !await IsBlocked(newX, newY, charWidth, charHeight))
173 {
174 character.TranslationX = newX;
175 character.TranslationY = newY;
176
178 if (character == jozio)
179 {
180 CheckMirrorCollection();
181 }
182 }
183 }
